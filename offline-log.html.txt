<script>
let logs = JSON.parse(localStorage.getItem('droneLogs'))||[];
const {jsPDF} = window.jspdf;
function renderLogs(){const b=document.getElementById('logList');b.innerHTML='';if(logs.length===0){b.innerHTML='<p style="color:#888">暂无记录</p>';return}logs.forEach((i,d)=>{const e=document.createElement('div');e.className='log-item';e.innerHTML=`<p><b>编号${d+1}</b> SN:${i.sn}</p><p>飞行员:${i.pilot}</p><p>地点:${i.location}</p><p>时间:${i.flyTime}</p><p>时长:${i.duration}分钟 | 空域:${i.space} | 任务:${i.task}</p><p>备注:${i.remark||'无'}</p><button class="btn btn-red" style="padding:5px 10px;font-size:13px" onclick="delLog(${d})">删除</button>`;b.appendChild(e)})}
function addLog(){const d={sn:document.getElementById('sn').value.trim(),pilot:document.getElementById('pilot').value.trim(),location:document.getElementById('location').value.trim(),flyTime:document.getElementById('flyTime').value,duration:document.getElementById('duration').value,space:document.getElementById('space').value,task:document.getElementById('task').value,remark:document.getElementById('remark').value.trim()};if(!d.sn||!d.pilot||!d.location||!d.flyTime||!d.duration){alert('关键信息不能为空');return}logs.push(d);localStorage.setItem('droneLogs',JSON.stringify(logs));renderLogs();clearForm();alert('保存成功')}
function delLog(d){logs.splice(d,1);localStorage.setItem('droneLogs',JSON.stringify(logs));renderLogs()}
function clearForm(){document.querySelectorAll('input,select,textarea').forEach(e=>e.value='')}
function clearAllLog(){if(confirm('确定清空所有日志？')){logs=[],localStorage.removeItem('droneLogs'),renderLogs()}}
function exportJSON(){const b=new Blob([JSON.stringify(logs,null,2)],{type:'application/json'});const a=document.createElement('a');a.href=URL.createObjectURL(b);a.download='无人机飞行日志.json';a.click()}
function exportPDF(){const d=new jsPDF();d.setFontSize(18);d.text('无人机飞行日志',14,22);d.setFontSize(11);let y=35;logs.forEach((i,d)=>{d.text(`【日志${d+1}】SN:${i.sn}`,14,y);y+=7;d.text(`飞行员:${i.pilot}`,14,y);y+=7;d.text(`地点:${i.location}`,14,y);y+=7;d.text(`时间:${i.flyTime} 时长:${i.duration}分钟`,14,y);y+=7;d.text(`空域:${i.space} 任务:${i.task}`,14,y);y+=7;d.text(`备注:${i.remark||'无'}`,14,y);y+=12;if(y>270){d.addPage();y=20}});d.save('无人机飞行日志.pdf')}
window.onload=renderLogs;
</script>
